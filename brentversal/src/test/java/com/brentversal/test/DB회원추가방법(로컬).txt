MemberAdd 파일은 로컬에서 관리자와 일반 사용자의 db를 입력하는 파일입니다.
로컬에서만 사용이 가능하며 기본적으로 mysql에 brentversal이 있어야 실행이 가능합니다.
workbench를 실행하고 localhost db에 접속 후 아래 sql문을 차례로 실행 한 뒤 클래스를 실행해야합니다.

-- 1) 데이터베이스 생성 (한글 저장 위해 utf8mb4)
CREATE DATABASE brentversal DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;

-- 2) 방금 만든 DB 선택
USE brentversal;

-- 3) 확인용 (아직 테이블 없음 = 정상)
SHOW TABLES;

-- 4) 실행 후 테이블 확인
USE brentversal;
SHOW TABLES;        -- 이제 members 가 보임
DESC members;       -- 컬럼 구조 확인

-- 5) member 데이터 확인
SELECT * FROM brentversal.members;
